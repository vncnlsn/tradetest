// ── CALENDAR ──────────────────────────────────────────────
function buildCalendar() {
  const grid = document.getElementById('calGrid');
  const startDay = 2; // April 1 2026 = Wednesday (0=Sun)
  const daysInMonth = 30;

  // April 2026 daily data
  const aprilData = {
    1:  { pnl: 100.00,  trades: 1 },
    2:  { pnl: 215.00,  trades: 1 },
    6:  { pnl: -54.30,  trades: 1 },
    7:  { pnl: 149.00,  trades: 1 },
    8:  { pnl: 223.50,  trades: 3 },
    9:  { pnl: 472.88,  trades: 6 },
    10: { pnl: 122.40,  trades: 2 },
  };

  // empty cells before start
  for (let i = 0; i < startDay; i++) {
    const el = document.createElement('div');
    el.className = 'cal-cell empty';
    grid.appendChild(el);
  }

  let weekPnl = 0;
  let dayOfWeek = startDay;

  for (let day = 1; day <= daysInMonth; day++) {
    const data = aprilData[day];
    const el = document.createElement('div');

    if (!data) {
      const isWeekend = (dayOfWeek % 7 === 0 || dayOfWeek % 7 === 6);
      el.className = 'cal-cell ' + (isWeekend ? 'empty' : 'no-trade');
      el.innerHTML = `<div class="cal-day-num">${day}</div>`;
    } else {
      const cls = data.pnl >= 0 ? 'profit-day' : 'loss-day';
      const sign = data.pnl >= 0 ? '+' : '';
      el.className = `cal-cell ${cls}`;
      el.innerHTML = `
        <div class="cal-day-num">${day}</div>
        <div>
          <div class="cal-day-pnl">${sign}$${Math.abs(data.pnl).toFixed(2)}</div>
          <div class="cal-day-trades">${data.trades} trade${data.trades > 1 ? 's' : ''}</div>
        </div>`;
      el.onclick = () => openDay(day, data);
      weekPnl += data.pnl;
    }

    grid.appendChild(el);
    dayOfWeek++;

    // weekly summary after Saturday
    if (dayOfWeek % 7 === 0 && day < daysInMonth) {
      const wk = document.createElement('div');
      wk.className = 'cal-week-summary';
      const sign = weekPnl >= 0 ? '+' : '';
      const cls  = weekPnl >= 0 ? 'profit' : 'loss';
      wk.innerHTML = `<span class="week-label">week total</span><span class="week-pnl ${cls}">${sign}$${weekPnl.toFixed(2)}</span>`;
      grid.appendChild(wk);
      weekPnl = 0;
    }
  }
}

function openDay(day, data) {
  const months = ['','Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  const days   = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
  const date   = new Date(2026, 3, day);
  const dname  = days[date.getDay()];

  document.getElementById('dayPanelTitle').textContent = `${dname}, April ${day}, 2026`;

  const pnl = data.pnl;
  const el  = document.getElementById('dayPanelPnl');
  el.textContent = (pnl >= 0 ? '+$' : '-$') + Math.abs(pnl).toFixed(2);
  el.className   = 'day-panel-pnl ' + (pnl >= 0 ? 'profit' : 'loss');

  // Build trades for this day
  const dateStr = `4/${day}`;
  const dayTrades = trades.filter(t => t.date === dateStr);
  const tbody = document.getElementById('dayTradesTbody');
  tbody.innerHTML = dayTrades.map(t => `
    <tr>

}

function initDrag() {

    w.addEventListener('dragover', e => {
      e.preventDefault();
      e.dataTransfer.dropEffect = 'move';
      if (w !== dragSrc) w.classList.add('drag-over');
    });
    w.addEventListener('dragleave', () => w.classList.remove('drag-over'));
    w.addEventListener('drop', e => {
      e.preventDefault();
      w.classList.remove('drag-over');
      if (dragSrc && dragSrc !== w) {
        const grid = document.getElementById('widgetGrid');
        const nodes = [...grid.children];
        const srcIdx = nodes.indexOf(dragSrc);
        const tgtIdx = nodes.indexOf(w);
        if (srcIdx < tgtIdx) grid.insertBefore(dragSrc, w.nextSibling);
        else grid.insertBefore(dragSrc, w);
      }
    });
  });
}

// ── EDIT MODE ─────────────────────────────────────────────
function toggleEditMode() {
  const grid = document.getElementById('widgetGrid');
  const bar  = document.getElementById('editBar');
  const btn  = document.getElementById('editToggle');
  const on   = !grid.classList.contains('edit-mode');

  grid.classList.toggle('edit-mode', on);
  bar.classList.toggle('active', on);
  btn.classList.toggle('active', on);
  btn.textContent = on ? '✓ done editing' : '⊞ customize';
}

function hideWidget(id) {
  const w = document.getElementById(id);
  if (w) w.style.display = 'none';
}

// ── CALENDAR ──────────────────────────────────────────────
function buildCalendar() {
  const grid = document.getElementById('calGrid');
  const startDay = 2; // April 1 2026 = Wednesday (0=Sun)
  const daysInMonth = 30;

  // April 2026 daily data
  const aprilData = {
    1:  { pnl: 100.00,  trades: 1 },
    2:  { pnl: 215.00,  trades: 1 },
    6:  { pnl: -54.30,  trades: 1 },
    7:  { pnl: 149.00,  trades: 1 },
    8:  { pnl: 223.50,  trades: 3 },
    9:  { pnl: 472.88,  trades: 6 },
    10: { pnl: 122.40,  trades: 2 },
  };

  // empty cells before start
  for (let i = 0; i < startDay; i++) {
    const el = document.createElement('div');
    el.className = 'cal-cell empty';
    grid.appendChild(el);
  }

  let weekPnl = 0;
  let dayOfWeek = startDay;

  for (let day = 1; day <= daysInMonth; day++) {
    const data = aprilData[day];
    const el = document.createElement('div');

    if (!data) {
      const isWeekend = (dayOfWeek % 7 === 0 || dayOfWeek % 7 === 6);
      el.className = 'cal-cell ' + (isWeekend ? 'empty' : 'no-trade');
      el.innerHTML = `<div class="cal-day-num">${day}</div>`;
    } else {
      const cls = data.pnl >= 0 ? 'profit-day' : 'loss-day';
      const sign = data.pnl >= 0 ? '+' : '';
      el.className = `cal-cell ${cls}`;
      el.innerHTML = `
        <div class="cal-day-num">${day}</div>
        <div>
          <div class="cal-day-pnl">${sign}$${Math.abs(data.pnl).toFixed(2)}</div>
          <div class="cal-day-trades">${data.trades} trade${data.trades > 1 ? 's' : ''}</div>
        </div>`;
      el.onclick = () => openDay(day, data);
      weekPnl += data.pnl;
    }

    grid.appendChild(el);
    dayOfWeek++;

    // weekly summary after Saturday
    if (dayOfWeek % 7 === 0 && day < daysInMonth) {
      const wk = document.createElement('div');
      wk.className = 'cal-week-summary';
      const sign = weekPnl >= 0 ? '+' : '';
      const cls  = weekPnl >= 0 ? 'profit' : 'loss';
      wk.innerHTML = `<span class="week-label">week total</span><span class="week-pnl ${cls}">${sign}$${weekPnl.toFixed(2)}</span>`;
      grid.appendChild(wk);
      weekPnl = 0;
    }
  }
}

function openDay(day, data) {
  const months = ['','Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  const days   = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
  const date   = new Date(2026, 3, day);
  const dname  = days[date.getDay()];

  document.getElementById('dayPanelTitle').textContent = `${dname}, April ${day}, 2026`;

  const pnl = data.pnl;
  const el  = document.getElementById('dayPanelPnl');
  el.textContent = (pnl >= 0 ? '+$' : '-$') + Math.abs(pnl).toFixed(2);
  el.className   = 'day-panel-pnl ' + (pnl >= 0 ? 'profit' : 'loss');

  // Build trades for this day
  const dateStr = `4/${day}`;
  const dayTrades = trades.filter(t => t.date === dateStr);
  const tbody = document.getElementById('dayTradesTbody');
  tbody.innerHTML = dayTrades.map(t => `
    <tr>
      <td class="ticker">${t.sym}</td>
      <td><span class="side-tag side-long">LONG</span></td>
      <td style="font-family:'DM Mono',monospace">$${t.entry.toFixed(4)}</td>
      <td style="font-family:'DM Mono',monospace">$${t.exit.toFixed(4)}</td>
      <td style="color:var(--muted)">${t.shares.toLocaleString()}</td>
      <td class="pnl-cell ${t.pnl >= 0 ? 'profit' : 'loss'}">${t.pnl >= 0 ? '+' : '-'}$${Math.abs(t.pnl).toFixed(2)}</td>
    </tr>`).join('');

  document.getElementById('dayPanel').classList.add('active');
  document.getElementById('dayPanel').scrollIntoView({ behavior: 'smooth', block: 'nearest' });
}

      <td><span class="side-tag side-long">LONG</span></td>
      <td style="font-family:'DM Mono',monospace">$${t.entry.toFixed(4)}</td>
      <td style="font-family:'DM Mono',monospace">$${t.exit.toFixed(4)}</td>
      <td style="color:var(--muted)">${t.shares.toLocaleString()}</td>
      <td class="pnl-cell ${t.pnl >= 0 ? 'profit' : 'loss'}">${t.pnl >= 0 ? '+' : '-'}$${Math.abs(t.pnl).toFixed(2)}</td>
      <td style="color:var(--muted)">—</td>
    </tr>`).join('');
}

// ── DRAG & DROP WIDGETS ───────────────────────────────────
let dragSrc = null;

document.addEventListener('DOMContentLoaded', () => {
  initDrag();
});

function initDrag() {
  document.querySelectorAll('.widget').forEach(w => {
    w.addEventListener('dragstart', e => {
      dragSrc = w;
      setTimeout(() => w.classList.add('dragging'), 0);
      e.dataTransfer.effectAllowed = 'move';
    });
    w.addEventListener('dragend', () => {
      w.classList.remove('dragging');
      document.querySelectorAll('.widget').forEach(x => x.classList.remove('drag-over'));
    });
    w.addEventListener('dragover', e => {
      e.preventDefault();
      e.dataTransfer.dropEffect = 'move';
      if (w !== dragSrc) w.classList.add('drag-over');
    });
    w.addEventListener('dragleave', () => w.classList.remove('drag-over'));
    w.addEventListener('drop', e => {
      e.preventDefault();
      w.classList.remove('drag-over');
      if (dragSrc && dragSrc !== w) {
        const grid = document.getElementById('widgetGrid');
        const nodes = [...grid.children];
        const srcIdx = nodes.indexOf(dragSrc);
        const tgtIdx = nodes.indexOf(w);
        if (srcIdx < tgtIdx) grid.insertBefore(dragSrc, w.nextSibling);
        else grid.insertBefore(dragSrc, w);
      }
    });
  });
}

// ── EDIT MODE ─────────────────────────────────────────────
function toggleEditMode() {
  const grid = document.getElementById('widgetGrid');
  const bar  = document.getElementById('editBar');
  const btn  = document.getElementById('editToggle');
  const on   = !grid.classList.contains('edit-mode');

  grid.classList.toggle('edit-mode', on);
  bar.classList.toggle('active', on);
  btn.classList.toggle('active', on);
  btn.textContent = on ? '✓ done editing' : '⊞ customize';
}

function hideWidget(id) {
  const w = document.getElementById(id);
  if (w) w.style.display = 'none';
}
